cmake_minimum_required(VERSION 3.22)
project(airpods LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
qt_add_executable(airpods src/main.cpp)
string(TIMESTAMP BUILD_DATE "%Y-%m-%d %H:%M:%S")
target_compile_definitions(airpods PRIVATE BUILD_DATE="${BUILD_DATE}")
target_include_directories(airpods PRIVATE src)
target_link_libraries(airpods PRIVATE Qt6::Widgets Qt6::Svg OpenSSL::Crypto)
if(DEFINED ENV{GIT_HASH})
    target_compile_definitions(airpods PRIVATE GIT_HASH="$ENV{GIT_HASH}")
endif()
if(WIN32)
    set(ZLIB_USE_STATIC_LIBS ON)
    find_package(ZLIB REQUIRED)
    target_link_libraries(airpods PRIVATE ZLIB::ZLIB)
    find_path(CPPWINRT_INCLUDE_DIR winrt/Windows.Devices.Bluetooth.Advertisement.h REQUIRED)
    target_include_directories(airpods SYSTEM PRIVATE "${CPPWINRT_INCLUDE_DIR}")
    target_compile_definitions(airpods PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(airpods PRIVATE windowsapp ws2_32 crypt32 bcrypt)
    if(MINGW)
        target_link_options(airpods PRIVATE -municode -static -static-libgcc -static-libstdc++)
        # Windows subsystem, so Explorer gives the tray no console of its own.
        # With -municode this is the wWinMain entry point main.cpp provides; the
        # console modes attach to the parent console instead.
        target_link_options(airpods PRIVATE -mwindows)
    endif()
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
    target_link_libraries(airpods PRIVATE PkgConfig::SYSTEMD)
endif()
if(NOT MSVC)
    target_compile_options(airpods PRIVATE -Wall -Wextra -Wshadow -Wnon-virtual-dtor)
endif()

# The Windows build runs through the same file with a MinGW toolchain, and
# nothing there can run the tests it would have built.
if(NOT CMAKE_CROSSCOMPILING)
    enable_testing()
    add_subdirectory(tests)
endif()
